# Runtime dependencies for the IAM control plane API.
#
# Installs cleanly on Windows, macOS and Linux — no system libraries required.
# The SAML stack needs native xmlsec and lives in requirements-saml.txt.
#
# Versions are pinned. Bump deliberately, never by accident.

fastapi==0.115.6
uvicorn[standard]==0.34.0
sqlalchemy[asyncio]==2.0.36
asyncpg==0.30.0
alembic==1.14.0
pydantic==2.10.4
pydantic-settings==2.7.0
python-json-logger==3.2.1

# Form parsing. The identity provider hands us a login by posting an HTML form to
# /saml/acs, and FastAPI refuses to start with a Form() parameter unless this is
# installed. Pure Python, so it goes in the cross-platform file.
python-multipart==0.0.20

# Reading and generating the keypair we sign assertions with (P5). Ships wheels
# for Windows, macOS and Linux, so it stays in the cross-platform file — which is
# the point: it means key handling is testable on a laptop and in CI, and only the
# XML signing itself needs the container. python3-saml pulls this in anyway; it is
# listed here so the version is pinned rather than inherited.
cryptography==44.0.0

# IANA timezone database for zoneinfo. Neither Windows nor python:*-slim ships
# one, so ZoneInfo("UTC") raises without this — which surfaced as Alembic
# failing to write a migration file.
tzdata==2026.3

# Outbound SCIM. Only the provisioning client uses it, and it was already a
# dev dependency for the test client — this promotes it to runtime because P6
# actually makes requests with it.
httpx==0.28.1
